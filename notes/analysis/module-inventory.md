scanning `src/main/java` of each module (34 buildable dirs; the 6 family/aggregator roots
have no `src/main/java`). The machine-readable copy lives in
`toolsmith/src/toolsmith/modules.py` (`ALIASES` + `PACKAGE_ROOTS`); this is its human-readable
mirror. **Do not maintain partial copies elsewhere - reference this file.**

## Regenerating

```bash
python - <<'PY'
import os
from pathlib import Path
ROOT = Path(os.environ.get("SIMPLIFIED_ROOT", "W:/Workspace/Java/Simplified"))
PRUNE = {"build",".gradle",".git",".idea","node_modules","cache","out","target","bin","gradle",".intellijPlatform"}
for dp, dn, fn in os.walk(ROOT):
    dn[:] = [d for d in dn if d not in PRUNE]
    if any(f.startswith("build.gradle") for f in fn):
        d = Path(dp); smj = d/"src"/"main"/"java"; pkg = "(no src/main/java)"
        if smj.is_dir():
            cur, parts = smj, []
            while True:
                subs = [x for x in cur.iterdir() if x.is_dir()]
                if len(subs) == 1 and not [x for x in cur.iterdir() if x.suffix == ".java"]:
                    parts.append(subs[0].name); cur = subs[0]
                else:
                    break
            pkg = ".".join(parts) or "(root/aggregator)"
        print(f"{d.relative_to(ROOT).as_posix():48}  {pkg}")
PY
```
